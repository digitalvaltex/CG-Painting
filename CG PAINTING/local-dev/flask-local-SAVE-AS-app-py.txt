@app.route("/")
def index():
    return send_from_directory(ROOT, "index.html")


@app.route("/<path:path>")
def static_file(path):
    if path == "send.php":
        return "", 404
    return send_from_directory(ROOT, path)


@app.route("/send.php", methods=["POST"])
def send_mail():
    naam = (request.form.get("naam") or "").strip()
    telefoon = (request.form.get("telefoon") or "").strip()
    email_addr = (request.form.get("email") or "").strip()
    postcode = (request.form.get("postcode") or "").strip()
    stad = (request.form.get("stad") or "").strip()
    type_werk = (request.form.get("type_werk") or "").strip()
    bericht = (request.form.get("bericht") or "").strip()

    type_werk_label = TYPE_LABELS.get(type_werk, type_werk)
    subject = f"Nieuw bericht van {naam}"
    reply_subject = "Re: " + subject
    reply_link = f"mailto:{email_addr}?subject={email.utils.quote(reply_subject)}" if email_addr else "#"

    telefoon_tel = re.sub(r"[^0-9+]", "", telefoon)
    if telefoon_tel and telefoon_tel[0] != "+":
        telefoon_tel = ("+32" + telefoon_tel[1:] if telefoon_tel.startswith("0") else "+32" + telefoon_tel)
    call_link = f"tel:{telefoon_tel}" if telefoon_tel else "#"

    calendar_title = f"Bel {naam} - CG Painting"
    calendar_details = f"Klant: {naam}\nTelefoon: {telefoon}\nE-mail: {email_addr}\nType werk: {type_werk_label}\n\nBericht:\n{bericht}"
    try:
        if ZoneInfo:
            tz = ZoneInfo("Europe/Brussels")
            tomorrow = (datetime.now(tz) + timedelta(days=1)).replace(hour=10, minute=0, second=0, microsecond=0)
            start_utc = tomorrow.astimezone(timezone.utc)
            end_utc = start_utc + timedelta(minutes=15)
            start_str = start_utc.strftime("%Y%m%dT%H%M%SZ")
            end_str = end_utc.strftime("%Y%m%dT%H%M%SZ")
            calendar_url = f"https://calendar.google.com/calendar/render?action=TEMPLATE&text={email.utils.quote(calendar_title)}&dates={start_str}/{end_str}&details={email.utils.quote(calendar_details)}"
        else:
            calendar_url = f"https://calendar.google.com/calendar/render?action=TEMPLATE&text={email.utils.quote(calendar_title)}&details={email.utils.quote(calendar_details)}"
    except Exception:
        calendar_url = f"https://calendar.google.com/calendar/render?action=TEMPLATE&text={email.utils.quote(calendar_title)}&details={email.utils.quote(calendar_details)}"

    html = build_html_email(
        naam, telefoon, email_addr, postcode, stad, type_werk, type_werk_label, bericht,
        reply_link, call_link, calendar_url
    )

    msg = MIMEMultipart("mixed")
    msg["Subject"] = subject
    msg["From"] = "Valtex Digital <no-reply@valtexdigital.com>"
    msg["To"] = TO_EMAIL
    msg["Reply-To"] = email_addr or TO_EMAIL

    msg.attach(MIMEText(html, "html", "utf-8"))

    if "fotos[]" in request.files:
        files = request.files.getlist("fotos[]")
    elif "fotos" in request.files:
        files = [request.files["fotos"]]
    else:
        files = []
    for f in files:
        if f and f.filename:
            part = MIMEBase("application", "octet-stream")
            part.set_payload(f.read())
            encoders.encode_base64(part)
            part.add_header("Content-Disposition", "attachment", filename=(f.filename or "foto"))
            msg.attach(part)

    gmail_user = os.environ.get("GMAIL_USER", TO_EMAIL)
    gmail_pass = os.environ.get("GMAIL_APP_PASSWORD")
    if not gmail_pass:
        return jsonify({"ok": False, "error": "GMAIL_APP_PASSWORD niet geconfigureerd. Zet .env of omgevingsvariabele."}), 500

    try:
        with smtplib.SMTP_SSL("smtp.gmail.com", 465) as s:
            s.login(gmail_user, gmail_pass)
            s.send_message(msg)
    except Exception as e:
        return jsonify({"ok": False, "error": "Verzenden mislukt: " + str(e)}), 500

    if request.headers.get("X-Requested-With") == "XMLHttpRequest":
        return jsonify({"ok": True, "message": "Bedankt! We nemen zo snel mogelijk contact op."})
    return redirect("/index.html#contact?sent=1", code=302)
